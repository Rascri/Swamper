/*******************************************************************************
  Heart_Rate_Display_Filtered_ESP32.pde  — CORREGIDO
  Fix 1: coeficientes del notch calculados en setup()
  Fix 2: BPM con ventana deslizante de 10 latidos + validación de rango
  BPM PROMEDIO DE 10 BEATS
  FILTRO NOTCH Y MEDIA MOVIL EN CODIGO
*******************************************************************************/

import processing.serial.*;

Serial myPort;

// ─── Gráfica ─────────────────────────────────────────────────────────────────
int   xPos       = 1;
float height_old = 0;
float height_new = 0;
float inByte     = 0;
float filtered   = 0;

// ─── BPM ─────────────────────────────────────────────────────────────────────
int     BPM            = 0;
int     beat_old       = 0;
int     beatCounter    = 0;
final int BEAT_WIN     = 10;           // ventana de 10 latidos (era 500 con zeros)
float[] beats          = new float[BEAT_WIN];
int     beatIndex      = 0;
int     beatCount      = 0;            // cuántos latidos válidos hay en el array
int     refractoryPeriod = 250;
int     lastBeatTime   = 0;
float   threshold      = 2600.0;
boolean belowThreshold = true;
boolean beatDetected = false;

// ─── Parámetros de filtros ────────────────────────────────────────────────────
final float FS      = 250.0;
final float ADC_MID = 2048.0;
final float ADC_MAX = 4095.0;

// Pasa-altos IIR (fc = 0.5 Hz) — calculado en setup()
float alpha_HP;
float hp_prev_x, hp_prev_y;

// Notch biquad 50 Hz (Audio EQ Cookbook) — calculado en setup()
// H(z) = (b0 + b1*z^-1 + b2*z^-2) / (1 + a1*z^-1 + a2*z^-2)
float nb0, nb1, nb2, na1, na2;
float notch_x1, notch_x2, notch_y1, notch_y2;

// Pasa-bajos IIR (fc = 40 Hz) — calculado en setup()
float alpha_LP;
float lp_prev_y;

// Media móvil FIR
final int MA_WIN    = 5;
float[]   ma_buffer = new float[MA_WIN];
int       ma_index  = 0;
float     ma_sum    = ADC_MID * MA_WIN;

// ─── Flags ───────────────────────────────────────────────────────────────────
boolean useHP  = true;
boolean useNOT = false;
boolean useLP  = true;
boolean useMA  = false;

PFont font;

// ─────────────────────────────────────────────────────────────────────────────
void setup() {
  size(1000, 400);
  printArray(Serial.list());
  myPort = new Serial(this, "COM3", 115200);   // ← ajustar puerto
  myPort.bufferUntil('\n');
  background(0xFF);
  font = createFont("Arial", 12, true);
  textFont(font);

  // ── Calcular coeficientes AQUÍ para que cos/sin estén disponibles ──────────

  // Pasa-altos 0.5 Hz
  float FC_HP = 0.5;
  alpha_HP = 1.0 / (1.0 + TWO_PI * FC_HP / FS);
  hp_prev_x = ADC_MID;
  hp_prev_y = ADC_MID;

  // Notch 50 Hz — fórmula Audio EQ Cookbook
  // w0 = 2π*f0/fs,  alpha = sin(w0)/(2*Q)
  float f0     = 50.0;
  float Q      = 30.0;
  float w0     = TWO_PI * f0 / FS;        // ≈ 1.2566 rad
  float alpha  = sin(w0) / (2.0 * Q);     // ≈ 0.01585
  float a0     = 1.0 + alpha;
  // Numerador
  nb0 =  1.0        / a0;
  nb1 = -2.0*cos(w0)/ a0;
  nb2 =  1.0        / a0;
  // Denominador (con signo negativo ya absorbido en la diferencia)
  na1 = -2.0*cos(w0)/ a0;                 // = nb1 (es el notch simétrico)
  na2 = (1.0-alpha) / a0;
  // Estados iniciales
  notch_x1 = ADC_MID; notch_x2 = ADC_MID;
  notch_y1 = ADC_MID; notch_y2 = ADC_MID;

  // Pasa-bajos 40 Hz
  float FC_LP = 40.0;
  alpha_LP  = (TWO_PI * FC_LP / FS) / (1.0 + TWO_PI * FC_LP / FS);
  lp_prev_y = ADC_MID;

  // Media móvil
  for (int i = 0; i < MA_WIN; i++) ma_buffer[i] = ADC_MID;
}

// ─────────────────────────────────────────────────────────────────────────────
void draw() {
  float display = map(filtered, 0, ADC_MAX, 0, height);
  height_new = height - display;
  line(xPos - 1, height_old, xPos, height_new);
  height_old = height_new;

  if (beatDetected) {
  stroke(0, 200, 0);
  line(xPos, 0, xPos, height);

  beatDetected = false;

  stroke(255, 0, 0);
}

  if (xPos >= width) {
    xPos = 0;
    background(0xFF);
    drawFilterStatus();
  } else {
    xPos++;
  }

  if (millis() % 128 == 0) {
    fill(0xFF);
    rect(0, 0, 260, 20);
    fill(0x00);
    textSize(13);
    text("BPM: " + BPM +
     "  Beats: " + beatCounter +
     "  TH: " + int(threshold),
     15, 14);
    drawFilterStatus();
  }
}

// ─────────────────────────────────────────────────────────────────────────────
void drawFilterStatus() {
  fill(0xFF);
  rect(0, height - 22, width, 22);
  fill(60);
  textSize(11);
  text("[H] HP:" + (useHP  ? "ON" : "--") +
       "  [N] Notch:" + (useNOT ? "ON" : "--") +
       "  [L] LP:"   + (useLP  ? "ON" : "--") +
       "  [M] MA:"   + (useMA  ? "ON" : "--") +
       "  [R]=RAW  [A]=ALL", 8, height - 7);
}

// ─────────────────────────────────────────────────────────────────────────────
void serialEvent(Serial myPort) {
  String inString = myPort.readStringUntil('\n');
  if (inString == null) return;
  inString = trim(inString);

  if (inString.equals("!")) {
    stroke(0, 0, 0xFF);
    inByte   = ADC_MID;
    filtered = ADC_MID;
    resetFilters();
  } else {
    stroke(0xFF, 0, 0);
    inByte   = float(inString);
    filtered = applyFilters(inByte);

if (filtered > threshold &&
    belowThreshold &&
    millis() - lastBeatTime > refractoryPeriod) {

  beatDetected = true;
  beatCounter++;

  println("BEAT #" + beatCounter +
          " ADC=" + int(filtered));

  lastBeatTime = millis();

  calculateBPM();

  belowThreshold = false;
}
else if (filtered < threshold) {
  belowThreshold = true;
}
  }
}

// ─────────────────────────────────────────────────────────────────────────────
float applyFilters(float x) {
  float y = x;

  if (useHP) {
    float hp_out = alpha_HP * (hp_prev_y + y - hp_prev_x);
    hp_prev_x = y;
    hp_prev_y = hp_out;
    y = hp_out + ADC_MID;
  }

  if (useNOT) {
    // Diferencia: y[n] = b0*x[n] + b1*x[n-1] + b2*x[n-2] - a1*y[n-1] - a2*y[n-2]
    float notch_out = nb0*y + nb1*notch_x1 + nb2*notch_x2
                      - na1*notch_y1 - na2*notch_y2;
    notch_x2 = notch_x1;  notch_x1 = y;
    notch_y2 = notch_y1;  notch_y1 = notch_out;
    y = notch_out;
  }

  if (useLP) {
    float lp_out = alpha_LP * y + (1.0 - alpha_LP) * lp_prev_y;
    lp_prev_y = lp_out;
    y = lp_out;
  }

  if (useMA) {
    ma_sum -= ma_buffer[ma_index];
    ma_buffer[ma_index] = y;
    ma_sum += y;
    ma_index = (ma_index + 1) % MA_WIN;
    y = ma_sum / MA_WIN;
  }

  return y;
}

// ─────────────────────────────────────────────────────────────────────────────
void resetFilters() {
  hp_prev_x = ADC_MID;  hp_prev_y = ADC_MID;
  notch_x1  = ADC_MID;  notch_x2  = ADC_MID;
  notch_y1  = ADC_MID;  notch_y2  = ADC_MID;
  lp_prev_y = ADC_MID;
  for (int i = 0; i < MA_WIN; i++) ma_buffer[i] = ADC_MID;
  ma_sum = ADC_MID * MA_WIN;
}

// ─────────────────────────────────────────────────────────────────────────────
void keyPressed() {
  switch (key) {
    case 'h': case 'H': useHP  = !useHP;  resetFilters(); break;
    case 'n': case 'N': useNOT = !useNOT; resetFilters(); break;
    case 'l': case 'L': useLP  = !useLP;  resetFilters(); break;
    case 'm': case 'M': useMA  = !useMA;  resetFilters(); break;
    case 'r': case 'R': useHP = useNOT = useLP = useMA = false; resetFilters(); break;
    case 'a': case 'A': useHP = useNOT = useLP = useMA = true;  resetFilters(); break;
    case '+':
  threshold += 50;
  println("Threshold = " + threshold);
  break;

case '-':
  threshold -= 50;
  println("Threshold = " + threshold);
  break;
  }
}

// ─────────────────────────────────────────────────────────────────────────────
void calculateBPM() {
  int beat_new = millis();
  int diff     = beat_new - beat_old;

  // Validar rango fisiológico: 30–200 BPM → 300ms–2000ms entre latidos
  if (diff < 300 || diff > 2000) {
    beat_old = beat_new;
    return;
  }

  float currentBPM    = 60000.0 / diff;
  beats[beatIndex]    = currentBPM;
  beatIndex           = (beatIndex + 1) % BEAT_WIN;
  if (beatCount < BEAT_WIN) beatCount++;   // contar solo slots válidos

  float total = 0;
  for (int i = 0; i < beatCount; i++) total += beats[i];
  BPM      = int(total / beatCount);       // promedio solo sobre datos reales
  beat_old = beat_new;
}
